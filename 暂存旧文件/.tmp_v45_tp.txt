    // [LMax Fix V5] 工业级延迟补种队列 V2：引入区块就绪检查与重试机制
    public static class DeferredQueue {
        private static class DeferredTask {
            String dimension;
            // [LMax Fix V40] 维度运行时本体：入队方直接提供 ResourceKey<Level>，消费侧零字符串解析 [长期记忆: 010]
            // 横杠化 dimension 字段保留，仅供 Data 文件路径序列化使用
            net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level> dim_key;
            ChunkPos chunk_pos;
            int retries;
            // [方向A重构] 新增：目标 chunk 和是否强制补写标记
            ChunkPos target_chunk; // 用于 PendingBlocks 补写，null 表示重新调 start
            boolean is_forced;     // true 表示只补写 PendingBlocks，不重新调 start
            
            // 旧版构造函数（重新调 start）
            DeferredTask(String d, net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level> dk, ChunkPos c) { // [LMax Fix V40] 新增dk [长期记忆: 010]
                this.dimension = d;
                this.dim_key = dk;
                this.chunk_pos = c;
                this.retries = 0;
                this.target_chunk = null;
                this.is_forced = false;
            }
            
            // [方向A重构] 新增构造函数（PendingBlocks 补写）
            DeferredTask(String d, net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level> dk, ChunkPos c, ChunkPos target, boolean forced) { // [LMax Fix V40] 新增dk [长期记忆: 010]
                this.dimension = d;
                this.dim_key = dk;
                this.chunk_pos = c;
                this.retries = 0;
                this.target_chunk = target;
                this.is_forced = forced;
            }
        }

        private static final java.util.concurrent.ConcurrentLinkedQueue<DeferredTask> queue = new java.util.concurrent.ConcurrentLinkedQueue<>();

        public static void add(String dimension, net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level> dim_key, ChunkPos chunk_pos) { // [LMax Fix V40] 新增dim_key [长期记忆: 010]
            // [LMax Fix V7] 容量上限检查，防止无界队列导致 OOM
            // [LMax Fix V42] 溢出驱逐升级见 evictOldest：旧实现无条件 poll 队头，把跨 chunk 树的 forced
            // 方块载荷与空数据 churn 任务一起混杀（世界22 随机空白区域根因之一，实测 184393 次溢出驱逐）
            while (queue.size() >= Handcode.Config.deferred_queue_max_size) {
                evictOldest();
            }
            queue.add(new DeferredTask(dimension, dim_key, chunk_pos)); // [LMax Fix V40]
        }

        // [方向A重构] 新增：PendingBlocks 补写任务
        public static void addForced(String dimension, net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level> dim_key, ChunkPos chunk_pos, ChunkPos target_chunk) { // [LMax Fix V40] 新增dim_key [长期记忆: 010]
            while (queue.size() >= Handcode.Config.deferred_queue_max_size) {
                evictOldest();
            }
            queue.add(new DeferredTask(dimension, dim_key, chunk_pos, target_chunk, true)); // [LMax Fix V40]
        }

        // [LMax Fix V42] 溢出驱逐：优先丢最老的非 forced 任务，队列全是 forced 时才丢绝对队头。
        // forced 任务是跨 chunk 树的方块载荷，丢一个 = 一块区域永远缺树；非 forced 只是重跑 start 的重试，
        // 丢了会被 TreeLocation 的 region 完成事件唤醒机制补回。
        // ConcurrentLinkedQueue 迭代弱一致：只读定位 + remove(对象) 原子删除，与并发 poll/add 安全共存
        // （remove 失败 = 该任务恰好被别的线程消费，外层 while 重查 size 即可）。
        private static void evictOldest () {
            DeferredTask oldest = null;
            for (DeferredTask t : queue) {
                if (t.is_forced == true) continue;
                oldest = t;
                break;
            }
            if (oldest != null) {
                queue.remove(oldest);
            } else {
                oldest = queue.poll(); // 兜底：队列全 forced，只能丢绝对队头
            }
            if (oldest != null) {
                // [LMax Fix V42] 溢出告警改走 log_queue_overflow 键控（旧 stderr 不受任何开关控制）
                if (Core.log_queue_overflow) System.err.println("[TansHugeTrees] DeferredQueue overflow, dropped task: " + oldest.dimension + " " + oldest.chunk_pos + " forced=" + oldest.is_forced);
            }
        }

        // [LMax Fix V43] P1 挂起机制 [长期记忆: 016]
        // 旧实现: 未就绪任务 retryList 回队重试 → 队列满载 evictOldest 互杀(V42 日志实锤 602,805 次,
        // FORCED/NORMAL DROPPED 均为 0 → 任务死于驱逐而非重试耗尽)。
        // 新架构: 未就绪任务挂起(不占队列/不轮询/不计数), 由 ChunkEvent.Load 反向唤醒(onChunkLoaded)。
        // 键 = task.chunk_pos(ready 检查中心); 加载区块 的 ±4 对角键即全部候选唤醒目标。
        private static final java.util.Map<ChunkPos, java.util.concurrent.ConcurrentLinkedQueue<DeferredTask>> suspended_tasks
                = new java.util.concurrent.ConcurrentHashMap<>();

        public static void processTick(net.minecraft.server.MinecraftServer server) {
            int processed = 0;
            DeferredTask task;

            // [LMax Fix V42] 每 tick 处理上限(配置项 deferred_queue_process_per_tick)防主线程 TPS 暴跌
            // [LMax Fix V43] 主线程瘦身 [长期记忆: 016]: 本方法运行于 eventTickServer(主线程), 只做
            // poll + 维度解析 + isReady(hasChunk×4 纯查询, 微秒级); 执行体(start→placeCalculate / placeForced)
            // 一律 submitExecute 进 TREE_GEN_EXECUTOR。旧实现执行体直接跑主线程 → 跑图期主线程替生成器
            // 等 IO + 数千方块树计算(V42 日志实锤 [Server thread] 打出 PendingBlocks.add 行 = 高频长卡顿+CPU空转根因)。
            // [LMax Fix V44.5] V43 注释的"一律 submitExecute"细化为双入口: processTick 主线程判定 PASS →
            // dispatchExecute 直通(判定已在主线程, 不再绕主线程队列一圈); 非主线程调用方(唤醒链)走
            // submitExecute 门. 执行体最终都进 TREE_GEN_EXECUTOR, 主线程职责不变. [长期记忆: 027]
            while (processed < Handcode.Config.deferred_queue_process_per_tick && (task = queue.poll()) != null) {
                processed++;

                // [LMax Fix V40] 维度串黑洞修复语义保留: 任务携带 ResourceKey<Level> 直查, 不反解析字符串 [长期记忆: 010]
                ServerLevel targetLevel = server.getLevel(task.dim_key);

                if (targetLevel == null) {
                    // [LMax Fix V40] 黑洞封口保留: 维度查找失败必须可见, 不静默丢弃
                    System.err.println("[TansHugeTrees] DeferredQueue dropped task: level not found for dim_key=" + task.dim_key + " chunk=" + task.chunk_pos + " forced=" + task.is_forced);
                    continue;
                }

                if (isReady(targetLevel, task)) {
                    // [LMax Fix V43] 就绪 → 提交 executor(tryExecute 内复查, 关闭"检查与执行间区块卸载"竞态)
                    // [LMax Fix V44.5] 上行 V43 所述复查已随 tryExecute 退役: 该复查跑在 executor 上, 被
                    // getChunkNow 线程卫兵恒判 false, 从未提供真实保护(V44.4 插桩裁决: 30 次主线程 PASS
                    // 任务全灭于此, [长期记忆: 027]). 本分支本就在主线程, PASS → dispatchExecute 直通执行.
                    dispatchExecute(server, task);
                } else {
                    // [LMax Fix V43] 不就绪 → 挂起(替代旧 retryList 回队: 602,805 次互杀的结构性根源)
                    suspendTask(targetLevel, task);
                }
            }

            // [LMax Fix V44] PlacementQueue 滴灌消费: 主线程按 getBudgetMs() 时间片落方块.
            // 挂点 = processTick(被 eventTickServer 每 tick 调用) = 主线程; 队列空时一次 isEmpty 零开销. [长期记忆: 018]
            PlacementQueue.drainTick();
            // [LMax Fix V44.3] 修2b 低水位回灌: 事件路径(Load 冲刷)因容量门被拒的滞留缓存方块在队列
            // 空闲时自愈重投. refillTick 自带 低水位/饱和/空缓存 三快退, 每 tick O(1). [长期记忆: 023]
            PendingBlocks.refillTick(server);
        }

        // [LMax Fix V43] ready 判定提取: ±4 对角 4 chunk 的 hasChunk + FULL
        // (原 processTick 内联逻辑原样提取, 语义不变; [执行代号33] 检查区块生成状态避免跨区块劈树)
        // [LMax Fix V44.3] 修3a: 判点集 4→9(dx+=8→dx+=4, 中心+四角+四边中点) + 提取 public isReadyChunk
        // 单一事实源. 旧四角版漏中心 proto 窗口: 角点已 FULL 而中心仍 proto 时挂起任务被唤醒放行,
        // 详细检测踩中心 proto 边 = getChunk(FULL) 强载残燃. 9 点 = 3a 总闸门判式的完整超集, 语义只收紧. [长期记忆: 023]
        // [LMax Fix V44.5] 主线程契约: 本方法依赖的 getChunkNow 在 1.20.1 ServerChunkCache 入口有
        // Thread.holdsLock(mainThread) 卫兵, 非主线程一律返回 null(V44.4 插桩裁决: 823 次证据B
        // hasChunk=true chunkNow=null 全为 executor 调用; 同任务同毫秒主线程 PASS / executor FAIL
        // 对质, [长期记忆: 027]). 因此本方法只允许在主线程调用(processTick / server.execute 门内),
        // 严禁 executor 直调; hasChunk 无罪——非主线程对已加载区块诚实返回 true.
        // V44.4 诊断插桩(证据A/B打印 + PASS 采样 + 计数器字段)随裁决完成拆除, 判定逻辑与插桩前等价.
        public static boolean isReadyChunk (ServerLevel level, ChunkPos chunk_pos) {
            for (int dx = -4; dx <= 4; dx += 4) {
                for (int dz = -4; dz <= 4; dz += 4) {
                    int cx = chunk_pos.x + dx;
                    int cz = chunk_pos.z + dz;
                    if (!level.getChunkSource().hasChunk(cx, cz)) {
                        return false;
                    }
                    // [LMax Fix V44] getChunk→getChunkNow: getChunk(x,z) 默认同步语义, 若该 chunk 处于
                    // proto→FULL 升格中会主动发起并 join 其 future(future 收尾依赖主线程 mainThreadProcessor
                    // 排队)——V43c 六线程卡 CompletableFuture.join 的直接机理. getChunkNow 为纯缓存查询,
                    // 未加载返回 null 零等待; hasChunk 已过滤未加载, 此处 null = proto 升格中, 同判 not ready. [长期记忆: 018]
                    // [LMax Fix V44.5] V44 时期把"非主线程返回 null"归因为"未加载"不完整——1.20.1 实测
                    // 还有主线程卫兵(见方法头契约): 非主线程即便已 FULL 也返回 null. 契约注释就是堵这个坑. [长期记忆: 027]
                    net.minecraft.world.level.chunk.ChunkAccess chunk = level.getChunkSource().getChunkNow(cx, cz);
                    if (chunk == null || !chunk.getHighestGeneratedStatus().isOrAfter(net.minecraft.world.level.chunk.ChunkStatus.FULL)) {
                        return false;
                    }
                }
            }
            return true;
        }

        private static boolean isReady(ServerLevel level, DeferredTask task) {
            return isReadyChunk(level, task.chunk_pos); // [LMax Fix V44.3] 判式上收 isReadyChunk, 单一事实源
        }

        // [LMax Fix V43] 统一提交入口: 所有执行路径(主线程 processTick / Load 唤醒 / 挂起复查)经此提交,
        // 保证执行体永远在 TREE_GEN_EXECUTOR, 主线程零重活。
        // [LMax Fix V44.5] 判定权收编主线程: 旧形态"投 executor → tryExecute 复查"把 isReady 的
        // getChunkNow 调到了 executor——线程卫兵下恒 false, 30 次主线程 PASS 任务全灭于此,
        // executeTask 零执行(V44.4 插桩裁决, 死因1 定罪点, [长期记忆: 027]). 新形态: server.execute
        // 把判定送回主线程, PASS → dispatchExecute 投 executor 跑重活, FAIL → 挂起链; tryExecute
        // 方法退役(职责并入本门). 已知残余竞态(接受): 主线程判定与 executor 执行间区块可能卸载——
        // 浪费一次计算, 无崩溃无状态损坏(PendingBlocks 三门自守); 旧 executor 复查在卫兵下从未
        // 提供真实保护, 拆除无损失.
        private static void submitExecute(net.minecraft.server.MinecraftServer server, DeferredTask task) {
            server.execute(() -> {
                try {
                    // [LMax Fix V44.5] 主线程判定门: server.execute 队列即主线程事件环, 此处 isReady 合法.
                    // server 关停后 execute 静默丢弃(仅关停窗口, 游戏内不可达).
                    ServerLevel level = server.getLevel(task.dim_key);
                    if (level == null) {
                        // [LMax Fix V40] 黑洞封口保留: 维度查找失败必须可见, 不静默丢弃
                        System.err.println("[TansHugeTrees] DeferredQueue dropped task: level not found for dim_key=" + task.dim_key + " chunk=" + task.chunk_pos + " forced=" + task.is_forced);
                        return;
                    }
                    if (isReady(level, task)) {
                        dispatchExecute(server, task);
                    } else {
                        suspendTask(level, task);
                    }
                } catch (Throwable t) {
                    System.err.println("[TansHugeTrees] submitExecute main-thread gate crashed (chunk=" + task.chunk_pos + " forced=" + task.is_forced + "):");
                    t.printStackTrace();
                }
            });
        }

        // [LMax Fix V44.5] 已判定直通执行: 唯一的"主线程判定 PASS → executor 执行"通道(processTick
        // 主线程分支 / submitExecute 门内 PASS / suspendTask recheck PASS 三处汇入). 调用方必须刚在
        // 本线程(主线程)完成 isReady PASS, 本方法不复核——在别的线程复核正是死因1 的教训. [长期记忆: 027]
        private static void dispatchExecute(net.minecraft.server.MinecraftServer server, DeferredTask task) {
            // [LMax Fix V44] 异常围栏: FutureTask 会捕获 lambda 内抛出的一切异常进 Future, 无人 read 的
            // Future = 异常黑洞(V43c ThreadingDetector permit 泄漏的无声载体). 围栏打印 stderr,
            // 此后执行路径异常不再无声蒸发. [长期记忆: 018]
            tannyjung.tanshugetrees_core.game.EventCenter.Server.TREE_GEN_EXECUTOR.submit(() -> {
                try {
                    executeTask(server, task);
                } catch (Throwable t) {
                    System.err.println("[TansHugeTrees] TREE_GEN_EXECUTOR task crashed (chunk=" + task.chunk_pos + " forced=" + task.is_forced + "):");
                    t.printStackTrace();
                }
            });
        }

        // [LMax Fix V43] 挂起: check-suspend-recheck 双检, 关闭"Load 事件在挂起前一刻触发"的竞态
        // (错过唤醒 → 任务永挂)。复查已就绪 → 经 submitExecute 提交(仍在 executor, 不占主线程); 否则入挂起表。
        // [LMax Fix V44.5] 语境更新: 本方法现仅被主线程同步调用(processTick FAIL 分支 / submitExecute
        // 门内 FAIL 分支), 调用方判定与 recheck 同线程零间隔——旧竞态窗口(executor 两检之间被 Load 插入)
        // 在主线程事件环顺序性下不复存在, recheck 保留为防御冗余(同步幂等零成本), 复查 PASS 改走
        // dispatchExecute 直通(旧 submitExecute 会把已判定任务再绕主线程队列一圈, [长期记忆: 027]).
        private static void suspendTask(ServerLevel level, DeferredTask task) {
            if (isReady(level, task)) {
                dispatchExecute(level.getServer(), task);
                return;
            }
            // [LMax Fix V43] 防御性兜底(配置项 deferred_queue_suspended_max): 正常玩家视距内挂起量级仅数百,
            // 触达上限 = 异常状态, 丢新并告警。
            if (suspended_tasks.size() >= Handcode.Config.deferred_queue_suspended_max) {
                if (Core.log_queue_overflow) System.err.println("[TansHugeTrees] suspended_tasks overflow, dropped: " + task.dimension + " " + task.chunk_pos + " forced=" + task.is_forced);
                return;
            }
            suspended_tasks.computeIfAbsent(task.chunk_pos, create -> new java.util.concurrent.ConcurrentLinkedQueue<>()).add(task);
            // [LMax Debug V43] 挂起诊断(键控 log_deferred_queue)
            if (Core.log_deferred_queue) System.out.println("[THT-DEBUG] suspendTask: " + task.chunk_pos + " forced=" + task.is_forced + " | suspended keys=" + suspended_tasks.size());
        }

        // [LMax Fix V43] Load 事件反向唤醒入口(EventCenter.eventChunkLoaded 调用) [长期记忆: 016]
        // 加载区块 的 ±4 对角 = 挂起任务的 ready 角点 → 4 个候选键, map.remove 原子摘除整个等待队列。
        // 幂等/并发安全: 两个角同时 Load 只有一方摘到(remove 原子性); 重复 Load 时任务已不在表内 → remove 返回 null。
        // [LMax Fix V44.3] 唤醒键 4→9(与 isReadyChunk 九点判式对称): 旧四角键漏中心/边中点——任务因中心
        // 未 FULL 挂起而四角已 FULL(不再 Load)时永挂; 中心升格的 Load 唤醒的是以本 chunk 为角的任务,
        // 不含以本 chunk 为中心的任务自身 → (0,0) 自唤醒键补此洞(修3a 挂起路径的生命线). [长期记忆: 023]
        public static void onChunkLoaded(ServerLevel level, ChunkPos loaded) {
            net.minecraft.server.MinecraftServer server = level.getServer();
            for (int dx = -4; dx <= 4; dx += 4) {
                for (int dz = -4; dz <= 4; dz += 4) {
                    java.util.concurrent.ConcurrentLinkedQueue<DeferredTask> waiting = suspended_tasks.remove(new ChunkPos(loaded.x + dx, loaded.z + dz));
                    if (waiting == null) continue;
                    for (DeferredTask task : waiting) {
                        // [LMax Fix V44.5] submitExecute = 主线程判定门(本方法运行于 executor, 门自动把判定送回主线程
                        // 后放行/挂起; 旧 tryExecute 的 executor 复查点已随其退役, [长期记忆: 027])
                        submitExecute(server, task);
                    }
                }
            }
        }

        // [LMax Fix V43] 任务执行体: 原 processTick 的就绪后分支原样迁移, 行为不变 [长期记忆: 016]
        private static void executeTask(net.minecraft.server.MinecraftServer server, DeferredTask task) {
            ServerLevel targetLevel = server.getLevel(task.dim_key);
            if (targetLevel == null) {
                System.err.println("[TansHugeTrees] DeferredQueue dropped task: level not found for dim_key=" + task.dim_key + " chunk=" + task.chunk_pos);
                return;
            }
            try {
                // [方向A重构] 区分两种任务: 重新调 start vs 只补写 PendingBlocks
                if (task.is_forced) {
                    // [LMax Debug V40] forced 任务诊断日志 [长期记忆: 009]
                    if (Core.log_deferred_queue) System.out.println("[THT-DEBUG] executeTask FORCED task: source=" + task.chunk_pos + " target=" + task.target_chunk);
                    // 区块已就绪, 强制补写 PendingBlocks
                    // [LMax Fix V44] placeForced 改投 PlacementQueue 滴灌: 返回值弃用, resync 由队列
                    // onComplete 触发(旧"forced_placed>0 立即 resync"时序已被取代) [长期记忆: 018]
                    // [LMax Fix V44.3] placeForced 并入 place(单一入口三门), 重载补写语义不变 [长期记忆: 023]
                    PendingBlocks.place(targetLevel, task.target_chunk);
                } else {
                    // [LMax Debug V40] 非 forced 任务诊断日志 [长期记忆: 009]
                    if (Core.log_deferred_queue) System.out.println("[THT-DEBUG] executeTask NORMAL task: chunk=" + task.chunk_pos);
                    // 旧版逻辑: 重新调 start
                    ChunkGenerator gen = targetLevel.getChunkSource().getGenerator();

    // [LMax Fix V42] 返回值语义：0=本次未写入任何方块（空数据且无 pending / 解析异常）；1=可能写入了方块（调用方应 resync）

    public static int start (LevelAccessor level_accessor, ServerLevel level_server, ChunkGenerator chunk_generator, String dimension, ChunkPos chunk_pos) {

        Core.GlobalLocking.test();

        // [LMax Fix V42] speculative register：读数据前先登记 churn 等待集 [长期记忆: 014]
        // 竞态防御：若先读后登记，可能错过"读取之后才完成的 region"的唤醒（T0读空→T1完成+wake未登记→
        // T2登记时邻region已全TRUE→误判终态漏树）。先登记保证任何后完成的 region 都能唤醒本 chunk；
        // 读到数据后立即注销（下方），等待集不会积累。
        TreeLocation.registerPendingEmpty(dimension, chunk_pos);
        ByteBuffer data = Data.get(dimension, chunk_pos);

        if (data.remaining() == 0) {
            // [LMax Debug V39] 诊断：空数据chunk是否也有缓存的方块（跨chunk树方块）
            if (Core.log_placer_start) System.out.println("[THT-DEBUG] TreePlacer.start() chunk " + chunk_pos + " EARLY RETURN (no tree data)");
            // [LMax Fix V39] 即使没有树数据，当前chunk可能被相邻chunk的树写入了方块，必须尝试place
            // [LMax Fix V42] place 现返回实际落块数：>0 说明本 chunk 有跨 chunk 方块落地，调用方需要 resync
            // [LMax Fix V44] place 改投 PlacementQueue 滴灌(void): 跨 chunk 方块的落块与 resync 均由
            // 队列负责, placed_pending 计数链废止 [长期记忆: 018]
            PendingBlocks.place(level_accessor, chunk_pos);
            // [LMax Fix V44.3] 空数据分支同步冲刷 litter(修4 的 start 侧入口): 旧路径只冲方块缓存, 本
            // chunk 无树数据但可能持有邻 chunk 树落的落叶缓存. 三门在 LeafLitterGeneration.place 内部,
            // proto 路径(Feature 钩子)留缓存由重载/回灌兜底. [长期记忆: 023]
            LeafLitterGeneration.place(level_accessor, level_server, chunk_generator, chunk_pos);
            // [LMax Fix V42] churn 斩杀：空数据不再 DeferredQueue.add 无限重入队（旧实现单日 64.6 万次
            // 空转重入，队列永久满载挤压真实 forced 载荷）。改为等待 TreeLocation 的 region 完成事件唤醒；
            // 3×3 邻 region 全部完成仍无数据 = 确定终态，注销退出（本 chunk 的树记录确定不存在）
            if (TreeLocation.allNeighborRegionsComplete(dimension, chunk_pos) == true) {
                TreeLocation.unregisterPendingEmpty(dimension, chunk_pos);
                if (Core.log_placer_start) System.out.println("[THT-DEBUG] EARLY RETURN chunk " + chunk_pos + " TERMINAL: 3x3 neighbor regions complete, no tree data (unregistered)");
            } else {
                if (Core.log_placer_start) System.out.println("[THT-DEBUG] EARLY RETURN chunk " + chunk_pos + " waiting for region wake");
            }
            // [LMax Fix V44] 返回 0: 方块落地异步化后本分支无同步落块证据(旧语义 placed_pending>0→1 废止)
            return 0;
        }

        // [LMax Fix V42] 读到数据 → 立即注销等待集，本 chunk 不再需要唤醒（speculative register 的对偶操作）
        TreeLocation.unregisterPendingEmpty(dimension, chunk_pos);

        // [LMax Debug] 追踪 TreePlacer 执行情况
        int data_size = data.remaining();
        long placer_start = System.currentTimeMillis();
        if (Core.log_placer_start) System.out.println("[THT-DEBUG] TreePlacer.start() chunk " + chunk_pos + " data: " + data_size + " bytes");

        String id = "";
        String chosen = "";
        int centerX = 0;
        int centerZ = 0;
        int from_chunkX = 0;
        int from_chunkZ = 0;
        int to_chunkX = 0;
        int to_chunkZ = 0;

        while (data.remaining() > 0) {

            // Get data
            {

                try {

                    id = CacheManager.getDictionary(String.valueOf(data.getShort()), true);
                    chosen = CacheManager.getDictionary(String.valueOf(data.getShort()), true);
                    centerX = data.getInt();
                    centerZ = data.getInt();
                    from_chunkX = data.getInt();
                    from_chunkZ = data.getInt();
                    to_chunkX = data.getInt();
                    to_chunkZ = data.getInt();

                } catch (Exception exception) {

                    OutsideUtils.exception(new Exception(), exception, "");
                        return 0; // [LMax Fix V42] 返回值语义：0=未写入任何方块（解析异常）

                }

            }

            DetailedDetection.test(level_accessor, level_server, chunk_generator, dimension, chunk_pos, from_chunkX, from_chunkZ, to_chunkX, to_chunkZ, id, chosen, centerX, centerZ);
        }

        // [LMax Fix V20] 解除主线程封印：让 PendingBlocks 回归异步线程直接写入
        // 彻底解决主线程被海量方块写入卡死，导致区块发包延迟和 TPS 归零的问题
        // [LMax Fix V20→V44 演进] V20 的"异步线程直写"在 V43c 实锤为死锁根因族(executor 碰 Level);
        // V44 改投 PlacementQueue: 主线程按 budget_ms 时间片滴灌, 开销恒定, 完工后 resync. [长期记忆: 018]
        PendingBlocks.place(level_accessor, chunk_pos);

        LeafLitterGeneration.place(level_accessor, level_server, chunk_generator, chunk_pos);

        // [LMax Debug] 追踪 TreePlacer 完成时间
        long placer_time = System.currentTimeMillis() - placer_start;
        if (placer_time > 100) {
            if (Core.log_placer_start) Core.logger.info("[THT-DEBUG] TreePlacer.start() chunk " + chunk_pos + " completed in " + placer_time + "ms (data was " + data_size + " bytes)");
        }
        // [LMax Fix V42] 返回值语义：1=进入了正常种树路径（保守视为可能写入了方块，调用方应 resync）
        return 1;
    }

    private static int[] getPartReduce (LevelAccessor level_accessor, String location, int centerX, int centerZ, int dead_tree_level) {

        RandomSource random = RandomSource.create(level_accessor.getServer().overworld().getSeed() ^ ((centerX * 341873128712L) + (centerZ * 132897987541L)));
        int count_trunk = 0;
        int count_bough = 0;
        int count_branch = 0;
        int count_limb = 0;
        int count_twig = 0;
        int count_sprig = 0;

        // Get Data
        {

            try {

                int[] data = Caches.TreeShape.getTreeShapeBlockCount(location);
                count_trunk = data[0];
                count_bough = data[1];
                count_branch = data[2];
                count_limb = data[3];
                count_twig = data[4];
                count_sprig = data[5];

            } catch (Exception exception) {

                OutsideUtils.exception(new Exception(), exception, "");
                return new int[0];

            }

        }

        // Convert Dead Level
        {

            if (dead_tree_level < 200) {

                dead_tree_level = dead_tree_level - 100;

            } else if (dead_tree_level < 300) {

                dead_tree_level = dead_tree_level - 200;

            } else if (dead_tree_level < 400) {

                dead_tree_level = dead_tree_level - 300;

            }

        }

        if (dead_tree_level >= 60) {

            // Only Trunk
            {

                count_bough = 0;
                count_branch = 0;
                count_limb = 0;
                count_twig = 0;
                count_sprig = 0;

                if (dead_tree_level == 60 || dead_tree_level == 70) {

                    count_trunk = (int) (Mth.nextDouble(random, 0.5, 1.0) * (double) count_trunk);

                } else if (dead_tree_level == 80 || dead_tree_level == 90) {

                    count_trunk = (int) (Mth.nextDouble(random, 0.1, 0.5) * (double) count_trunk);

                }

            }

        } else {

            // General
            {

                if (dead_tree_level == 50) {

                    count_bough = (int) (random.nextDouble() * count_bough);

                }

                if (dead_tree_level < 50) {

                    if (dead_tree_level == 40) {

                        count_branch = (int) (random.nextDouble() * count_branch);

                    }

                } else {

                    count_branch = 0;

                }

                if (dead_tree_level < 40) {

                    if (dead_tree_level == 30 && count_limb > 0) {

                        count_limb = (int) (random.nextDouble() * count_limb);

                    }

                } else {

                    count_limb = 0;

                }

                if (dead_tree_level < 30) {

                    if (dead_tree_level == 20 && count_twig > 0) {

                        count_twig = (int) (random.nextDouble() * count_twig);

                    }

                } else {

                    count_twig = 0;

                }

                if (dead_tree_level < 20) {

                    if (dead_tree_level == 10 && count_sprig > 0) {

                        count_sprig = (int) (random.nextDouble() * count_sprig);

                    }

                } else {

                    count_sprig = 0;

                }

            }

        }

        return new int[]{count_trunk, count_bough, count_branch, count_limb, count_twig, count_sprig};

    }

                                if (random.nextDouble() < Handcode.Config.tree_decorations_normal_chance) {

                                    if (tree_decoration_normal.isEmpty() == false) {

                                        // [方向A重构] 用方块所在 chunk
                                        Function.add(new ChunkPos(pos), pos, tree_decoration_normal.get(random.nextInt(tree_decoration_normal.size())));

                                    }

                                }

                            }

                            if (dead_tree_level > 0) {

                                if (random.nextDouble() < Handcode.Config.tree_decorations_decay_chance) {

                                    if (tree_decoration_decay.isEmpty() == false) {

                                        // [方向A重构] 用方块所在 chunk
                                        Function.add(new ChunkPos(pos), pos, tree_decoration_decay.get(random.nextInt(tree_decoration_decay.size())));

                                    }

                                }

                            }

                        }

                    }

                    // Summon Marker
                    {

                        // At Center
                        if (posX == 0 && posY == 0 && posZ == 0) {

                            if (Handcode.Config.tree_location == true && dead_tree_level == 0) {

                                if (can_leaves_decay == true || can_leaves_drop == true || can_leaves_regrow == true) {

                                    String marker_data = "{ForgeData:{tanshugetrees:{file:\"" + location + "\",tree_settings:\"" + path_settings + "\",rotation:" + rotation_mirrored[0] + ",mirrored:" + rotation_mirrored[1] + "}}}";
                                    GameUtils.Mob.summonWorldGen(level_server, pos.getCenter(), "minecraft:marker", id, "TANSHUGETREES-tree_location", marker_data);

                                }

                            }

                        }

                    }

                    can_run_function = true;

                } else {

                    // Function
                    {
                        // Separate because start and end function no need to test "can run here?"
                        if (can_run_function == true || type == 210 || type == 220) {

                            function = functions.get(type);

                            if (function == null) {

                                continue;

                            }

                            // [方向A重构] 用方块所在 chunk
                            Function.add(new ChunkPos(pos), pos, function);

                        }

                    }

                }

            }

        }

    }

    public static class Data {

        // [LMax Fix V3]彻底废除同步等待，改用 CompletableFuture 异步加载。
        // 任何线程请求数据时，如果未加载完成，直接返回空，绝不阻塞当前线程。
        // 彻底根除 DH 线程池排队、主线程级联卡死的问题。
        private static final Map<String, java.util.concurrent.CompletableFuture<Map<ChunkPos, ByteArrayOutputStream>>> bin_convert_futures = new java.util.concurrent.ConcurrentHashMap<>();

        public static void clear () {
            bin_convert_futures.clear();
        }
        // [LMax Fix V41] 负缓存失效入口 [长期记忆: 012]
        // 根因：get() 的 computeIfAbsent 将“region 文件尚不存在”这一瞬时状态解析为空 map 并永久缓存
        // （本会话 region key 数远小于 256 淘汰阈值，且无任何失效路径），数据落盘后读方仍命中空结果
        // → 25442/25442 全空读 → 零树。修复：生产者 TreeLocation.flushCachesAsync 落盘后调用本方法
        // 失效对应 region 的 future；消费方（DeferredQueue 400tick 重试节奏）下次 get 重新读盘。事件驱动零轮询。
        public static void invalidate (String dimension, int regionX, int regionZ) {
            bin_convert_futures.remove(dimension + "/" + regionX + "," + regionZ);
        }

        private static void clearChunk (String dimension, ChunkPos chunk_pos) {
            // 异步模式下，clearChunk 操作变得复杂且无必要，直接清空整个 Future 缓存即可。
        }

        private static ByteBuffer get (String dimension, ChunkPos chunk_pos) {
            int regionX = chunk_pos.x >> 5;
            int regionZ = chunk_pos.z >> 5;

            String key = dimension + "/" + regionX + "," + regionZ;

            // [LMax Fix V7] 容量上限淘汰，防止 bin_convert_futures 无限增长导致 OOM
            if (!bin_convert_futures.containsKey(key)) {
                while (bin_convert_futures.size() >= Handcode.Config.bin_convert_futures_max_entries) {
                    java.util.Iterator<Map.Entry<String, java.util.concurrent.CompletableFuture<Map<ChunkPos, ByteArrayOutputStream>>>> it = bin_convert_futures.entrySet().iterator();
                    if (it.hasNext()) {
                        it.next();
                        it.remove();
                    } else {
                        break;
                    }
                }
            }

            java.util.concurrent.CompletableFuture<Map<ChunkPos, ByteArrayOutputStream>> future = bin_convert_futures.computeIfAbsent(key, k ->
                java.util.concurrent.CompletableFuture.supplyAsync(() -> {
                    Map<ChunkPos, ByteArrayOutputStream> regionData = new HashMap<>();
                    ByteBuffer bin = FileManager.readBIN(Core.path_world_mod + "/world_gen/place/" + k + ".bin");
                    if (bin != null) {
                        ByteArrayOutputStream stream = new ByteArrayOutputStream();
                        int from_chunkX = 0;
                        int from_chunkZ = 0;
                        int to_chunkX = 0;
                        int to_chunkZ = 0;
                        short id = 0;
                        short chosen = 0;
                        int centerX = 0;
                        int centerZ = 0;

                        while (bin.remaining() > 0) {
                            try {
                                id = bin.getShort();
                                chosen = bin.getShort();
                                centerX = bin.getInt();
                                centerZ = bin.getInt();
                                from_chunkX = bin.getInt();
                                from_chunkZ = bin.getInt();
                                to_chunkX = bin.getInt();
                                to_chunkZ = bin.getInt();

                                stream.reset();
                                stream.write(OutsideUtils.Data.convertShortToArrayByte(id));
                                stream.write(OutsideUtils.Data.convertShortToArrayByte(chosen));
                                stream.write(OutsideUtils.Data.convertIntToArrayByte(centerX));
                                stream.write(OutsideUtils.Data.convertIntToArrayByte(centerZ));
                                stream.write(OutsideUtils.Data.convertIntToArrayByte(from_chunkX));
                                stream.write(OutsideUtils.Data.convertIntToArrayByte(from_chunkZ));
                                stream.write(OutsideUtils.Data.convertIntToArrayByte(to_chunkX));
                                stream.write(OutsideUtils.Data.convertIntToArrayByte(to_chunkZ));

                                for (int scanX = from_chunkX; scanX <= to_chunkX; scanX++) {
                                    for (int scanZ = from_chunkZ; scanZ <= to_chunkZ; scanZ++) {
                                        if (regionX == scanX >> 5 && regionZ == scanZ >> 5) {
                                            regionData.computeIfAbsent(new ChunkPos(scanX, scanZ), create -> new ByteArrayOutputStream()).write(stream.toByteArray());
                                        }
                                    }
                                }
                            } catch (Exception exception) {
                                OutsideUtils.exception(new Exception(), exception, "");
                                break;
                            }
                        }
                    }
                    return regionData;
                })
            );

            // 核心妥协：如果 Future 还没完成，当前线程绝对不等待，直接返回空数据放弃本次生成。
            // 依靠 TreeLocation 中的 +-4 Chunk 偏移重复生成逻辑来补偿漏掉的树木。
            if (future.isDone()) {
                try {
                    Map<ChunkPos, ByteArrayOutputStream> data = future.get();
                    ByteArrayOutputStream stream = data.get(chunk_pos);
                    if (stream == null) {
                        return ByteBuffer.allocate(0);
                    } else {
                        return ByteBuffer.wrap(stream.toByteArray());
                    }
                } catch (Exception e) {
                    return ByteBuffer.allocate(0);
                }
            } else {
                return ByteBuffer.allocate(0);
            }
        }

    }

                        if (size >= 0) {

                            // [P2 修复] 限制扫描 chunk 数量，防止大范围扫描导致卡顿
                            int chunk_scan_count = 0;
                            boolean chunk_scan_limit_reached = false;

                            for (int scanX = from_chunkX - size; scanX <= to_chunkX + size; scanX++) {

                                    if (chunk_scan_count >= Handcode.Config.structure_detection_max_chunks) {

                                        chunk_scan_limit_reached = true;
                                        break;

                                    }

                                    chunk_scan_count = chunk_scan_count + 1;
                                    chunk_pos_test = new ChunkPos(scanX, scanZ);

                                      if (GameUtils.Space.testChunkStatus(level_accessor, chunk_pos_test, "structure_references") == true) {

                                          // [LMax Fix V44.3] 修3b 语义域分流: getChunk(x,z) 双重人格 API——
                                          // WorldGenRegion 接收者返回区域内 proto chunk(worldgen 管线语义合法);
                                          // ServerLevel 接收者 = getChunk(FULL, create=true) 同步强制加载 + join
                                          // 升格 future(收尾依赖主线程 mainThreadProcessor = V43c 跨线程死锁机理),
                                          // 且被强载 chunk 触发 Load→+5s→start→更远扫描 = 雪球每跳 64 格的公共根.
                                          // ServerLevel 侧改 getChunkNow 纯缓存查询(未加载/proto 升格中 null 零等待):
                                          // null → 空引用集放行(未加载 = 无结构记录, 与原 references 判空跳过语义等价). [长期记忆: 023]
                                          Map<Structure, LongSet> chunk_references;
                                          if (level_accessor instanceof net.minecraft.server.level.ServerLevel) {
                                              net.minecraft.world.level.chunk.ChunkAccess chunk_now =
                                                      ((net.minecraft.server.level.ServerLevel) level_accessor).getChunkSource().getChunkNow(chunk_pos_test.x, chunk_pos_test.z);
                                              chunk_references = chunk_now == null ? java.util.Collections.emptyMap() : chunk_now.getAllReferences();
                                          } else {
                                              // WorldGenRegion(worldgen 管线/proto): 原语义原样保留
                                              chunk_references = level_accessor.getChunk(chunk_pos_test.x, chunk_pos_test.z).getAllReferences();
                                          }
                                          references = chunk_references;

                                        if (references.size() > 0) {

                                            for (Structure structure : references.keySet()) {

                                                if (structure.step() == GenerationStep.Decoration.SURFACE_STRUCTURES) {

                                                    break test;

                                                }

                                            }

                                        }

                                    }

                                }

                            }

                        }

                    }

                    if (sizeX != 0 || sizeZ != 0) {

                        if (testSurfaceSmoothness(level_accessor, level_server, chunk_generator, pos_center, sizeX, sizeY, sizeZ, center_sizeX, center_sizeY, center_sizeZ, pos_original) == false) {

                            break test;

                        }

                        if (dead_tree_level > 200) {

                            if (testFallenArea(level_accessor, level_server, chunk_generator, location, pos_center, rotation_mirrored, fallen_direction, dead_tree_level) == false) {

                                break test;

                            }

                        }

                    }

                    pass = true;

                }

            }

        

          
            if (pass == true) {

                long pc_start = System.currentTimeMillis();
                // [方向A重构] 调用 placeCalculate() 替代旧 place()
                placeCalculate(level_accessor, level_server, chunk_pos, id, location, path_settings, pos_center, rotation_mirrored, dead_tree_level, fallen_direction);

                }

                // 修复：全部走 DeferredQueue，不在生成期间直接访问邻近区块。
                for (int scanX = from_chunkX; scanX <= to_chunkX; scanX++) {
                    for (int scanZ = from_chunkZ; scanZ <= to_chunkZ; scanZ++) {

                        ChunkPos target_cp = new ChunkPos(scanX, scanZ);

                        if (target_cp.equals(chunk_pos)) {
                            continue;
                        }

                        DeferredQueue.addForced(dimension, level_server.dimension(), chunk_pos, target_cp); // [LMax Fix V40] 传递维度key修复串黑洞 [长期记忆: 010]

                    }
                }

            }

        }

        private static boolean testSurfaceSmoothness (LevelAccessor level_accessor, ServerLevel level_server, ChunkGenerator chunk_generator, BlockPos pos_center, int sizeX, int sizeY, int sizeZ, int center_sizeX, int center_sizeY, int center_sizeZ, BlockPos pos_original) {

            if (Handcode.Config.surface_smoothness_detection == true) {

                int test_sizeX = sizeX - center_sizeX;
                int test_sizeZ = sizeZ - center_sizeZ;
                int test_center_sizeX = center_sizeX;
                int test_center_sizeZ = center_sizeZ;

                int pos1 = GameUtils.Space.getHeightWorldGen(level_accessor, level_server, chunk_generator, pos_center.getX() - test_center_sizeX, pos_center.getZ() - test_center_sizeZ, "OCEAN_FLOOR", "OCEAN_FLOOR_WG");
                int pos2 = GameUtils.Space.getHeightWorldGen(level_accessor, level_server, chunk_generator, pos_center.getX() - test_center_sizeX, pos_center.getZ() + test_sizeZ, "OCEAN_FLOOR", "OCEAN_FLOOR_WG");
                int pos3 = GameUtils.Space.getHeightWorldGen(level_accessor, level_server, chunk_generator, pos_center.getX() + test_sizeX, pos_center.getZ() - test_center_sizeZ, "OCEAN_FLOOR", "OCEAN_FLOOR_WG");
                int pos4 = GameUtils.Space.getHeightWorldGen(level_accessor, level_server, chunk_generator, pos_center.getX() + test_sizeX, pos_center.getZ() + test_sizeZ, "OCEAN_FLOOR", "OCEAN_FLOOR_WG");

                int height_up = (sizeY - center_sizeY) + Math.abs(pos_center.getY() - pos_original.getY());
                height_up = pos_original.getY() + (int) Math.ceil(height_up * Handcode.Config.surface_smoothness_detection_height_up * 0.01);
                int height_down = center_sizeY + Math.abs(pos_center.getY() - pos_original.getY());
                height_down = pos_original.getY() - (int) Math.ceil(height_down * Handcode.Config.surface_smoothness_detection_height_down * 0.01);
                boolean test1 = (pos_original.getY() < pos1 && height_up > pos1) || (pos_original.getY() >= pos1 && pos1 > height_down);
                boolean test2 = (pos_original.getY() < pos2 && height_up > pos2) || (pos_original.getY() >= pos2 && pos2 > height_down);
                boolean test3 = (pos_original.getY() < pos3 && height_up > pos3) || (pos_original.getY() >= pos3 && pos3 > height_down);
                boolean test4 = (pos_original.getY() < pos4 && height_up > pos4) || (pos_original.getY() >= pos4 && pos4 > height_down);

                return test1 == true && test2 == true && test3 == true && test4 == true;

            }

            return true;

        }

                    reduce_sprig = data[5];

                } catch (Exception exception) {

            }

            int left_before_test = 0;

            // Get Left
            {

                double total = reduce_trunk + reduce_bough + reduce_branch + reduce_limb + reduce_twig + reduce_sprig;
                left_before_test = (int) Math.ceil(total * 0.5);

            }

            boolean is_only_trunk = dead_tree_level >= 60;
            int distance_skip = (int) Math.floor(left_before_test / 16.0);
        

          
            int distance_skip_test = 0;
            // [P1 修复] 高度检查计数器，用于限制 getHeightWorldGen 调用次数
            int height_check_count = 0;
            BlockPos pos = null;

            int loop = 0;
            short type = 0;
            short posX = 0;
            short posY = 0;
            short posZ = 0;

            for (short scan : Caches.TreeShape.getTreeShapeData(location)) {

                // Get Data
                {

                    loop = loop + 1;

                    if (loop == 1) {

                        type = scan;

                    } else if (loop == 2) {

                        posX = scan;

                    } else if (loop == 3) {

                        posY = scan;

                    } else {

                        posZ = scan;
                        loop = 0;

                    }

                }

                if (loop == 0) {

                    if (OutsideUtils.Mathematics.isNumberStartWith(type, 1) == true) {

                        // Skip Roots
                        {

                            if (OutsideUtils.Mathematics.isNumberStartWith(type, 110) == true || OutsideUtils.Mathematics.isNumberStartWith(type, 111) == true || OutsideUtils.Mathematics.isNumberStartWith(type, 112) == true || OutsideUtils.Mathematics.isNumberStartWith(type, 113) == true) {

                                continue;

                            }

                        }

                        // Dead Tree Reduction
                        {

                            // Basic Style
                            {

                                if (OutsideUtils.Mathematics.isNumberStartWith(type, 120) == true) {

                                    continue;

                                } else if (OutsideUtils.Mathematics.isNumberStartWith(type, 119) == true) {

                                    if (reduce_sprig > 0) {

                                        reduce_sprig = reduce_sprig - 1;

                                    } else {

                                        continue;

                                    }

                                } else if (OutsideUtils.Mathematics.isNumberStartWith(type, 118) == true) {

                                    if (reduce_sprig == 0) {

                                        if (reduce_twig > 0) {

                                            reduce_twig = reduce_twig - 1;

                                        } else {

                                            continue;

                                        }

                                    }

                                } else if (OutsideUtils.Mathematics.isNumberStartWith(type, 117) == true) {

                                    if (reduce_twig == 0) {

                                        if (reduce_limb > 0) {

                                            reduce_limb = reduce_limb - 1;

                                        } else {

                                            continue;

                                        }

                                    }

                                } else if (OutsideUtils.Mathematics.isNumberStartWith(type, 116) == true) {

                                    if (reduce_limb == 0) {

                                        if (reduce_branch > 0) {

                                            reduce_branch = reduce_branch - 1;

                                        } else {

                                            continue;

                                        }

                                    }

                                } else if (OutsideUtils.Mathematics.isNumberStartWith(type, 115) == true) {

                                    if (reduce_branch == 0) {

                                        if (reduce_bough > 0) {

                                            reduce_bough = reduce_bough - 1;

                                        } else {

                                            continue;

                                        }

                                    }

                                }

                            }

                            if (is_only_trunk == true) {

                                // Only Trunk
                                {

                                    if (OutsideUtils.Mathematics.isNumberStartWith(type, 114) == true) {

                                        if (reduce_trunk > 0) {

                                            reduce_trunk = reduce_trunk - 1;

                                        } else {

                                            continue;

                                        }

                                    }

                                }

                            }

                        }

                        if (left_before_test > 0) {

                            left_before_test = left_before_test - 1;

                        } else {

                            // Distance Skip
                            {

                                if (distance_skip > 0) {

                                    if (distance_skip_test > 0) {

                                        distance_skip_test = distance_skip_test - 1;
                                        continue;

                                    } else {

                                        distance_skip_test = distance_skip;

                                    }

                                }

                            }

        

          
                            pos = new BlockPos(posX, posY, posZ);
                            pos = OutsideUtils.convertPosRotationMirrored(pos, rotation_mirrored);
                            pos = OutsideUtils.convertPosFallen(pos, fallen_direction);
                            pos = pos.offset(pos_center.getX(), pos_center.getY(), pos_center.getZ());

                            // [P1 修复] 超过最大高度检查次数时停止检测，避免大树形状导致大量噪声计算卡顿
                            if (height_check_count >= Handcode.Config.test_fallen_area_max_height_checks) {

                                return false;

                            }

                            height_check_count = height_check_count + 1;

                            if (pos.getY() <= GameUtils.Space.getHeightWorldGen(level_accessor, level_server, chunk_generator, pos.getX(), pos.getZ(), "OCEAN_FLOOR_WG", "OCEAN_FLOOR_WG")) {

                                return true;

                            }

                        }

                    }

                }

            }

            return false;

        }

    }

    private static class LeafLitterGeneration {

        // [LMax Fix V7] 替换为线程安全容器，解决 ForkJoinPool 并发修改异常
        private static final Map<ChunkPos, Map<BlockPos, BlockState>> cache_locations = new java.util.concurrent.ConcurrentHashMap<>();

        private static void add (ChunkPos chunk_pos, BlockPos pos, BlockState block) {
            cache_locations.computeIfAbsent(chunk_pos, create -> new java.util.concurrent.ConcurrentHashMap<>()).put(pos, block);
        }

        // [LMax Fix V44] place 整体改投 PlacementQueue: 原实现在调用线程(=TREE_GEN_EXECUTOR)上直接做
        // getHeightWorldGen + LeafLitter.create 写世界, 属于"executor 碰 Level"违例(V43c 死锁根因族).
        // 现在原子摘取缓存后入队, 高度查询+落叶创建作为 placer 策略在主线程滴灌执行,
        // 完工后 onComplete 触发 resync(方块落地后才发包). [长期记忆: 018]
        private static void  place (LevelAccessor level_accessor, ServerLevel level_server, ChunkGenerator chunk_generator, ChunkPos chunk_pos) {

            // [LMax Fix V44.3] 修1 同款三门: ①proto 分流(level_accessor instanceof, worldgen 管线留缓存)
            // ②FULL 门(getChunkNow+isOrAfter, level_server 直查零等待) ③容量门(isSaturated).
            // 拒绝 = 留缓存零丢失, 由 Load 路径 start(L358)/重载 flushPendingLitter(修4)后续冲刷. [长期记忆: 023]
            if ((level_accessor instanceof net.minecraft.server.level.ServerLevel) == false) {
                return;
            }
            net.minecraft.world.level.chunk.ChunkAccess chunk_now =
                    level_server.getChunkSource().getChunkNow(chunk_pos.x, chunk_pos.z);
            if (chunk_now == null || !chunk_now.getHighestGeneratedStatus().isOrAfter(net.minecraft.world.level.chunk.ChunkStatus.FULL)) {
                return;
            }
            if (PlacementQueue.isSaturated()) {
                return;
            }
            // [LMax Fix V44] get→remove 原子摘取(所有权转移给队列): 并发调用天然幂等, 后到者拿 null 零开销
            Map<BlockPos, BlockState> data = cache_locations.remove(chunk_pos);

            if (data == null) {
                return;
            }

            // [LMax Fix V44] 落叶策略: 每块查地表高度(MOTION_BLOCKING_NO_LEAVES)后 LeafLitter.create,
            // 逻辑与原实现逐行等价, 仅执行线程从 executor 换到主线程滴灌
            java.util.function.BiConsumer<BlockPos, BlockState> placer = (pos, block) -> {

                int height_motion = GameUtils.Space.getHeightWorldGen(level_accessor, level_server, chunk_generator, pos.getX(), pos.getZ(), "MOTION_BLOCKING_NO_LEAVES", "WORLD_SURFACE_WG");

                if (height_motion < pos.getY()) {

                    LeafLitter.create(level_accessor, level_server, pos.atY(height_motion), block, false);

                }

            };

            PlacementQueue.enqueue(level_server, chunk_pos, data, placer,
                    () -> tannyjung.tanshugetrees_core.game.EventCenter.Server.resyncChunk(level_server, chunk_pos));
        }

    }

    // [方向A重构] 新增 PendingBlocks 类：按 chunk 分组缓存方块，解决跨 chunk 写入时序问题
    private static class PendingBlocks {

        // 线程安全容器：ChunkPos -> (BlockPos -> BlockState)
        private static final Map<ChunkPos, Map<BlockPos, BlockState>> cache_blocks = new java.util.concurrent.ConcurrentHashMap<>();

        // [LMax Debug V39] 诊断计数器：追踪入缓存方块总数
        private static final java.util.concurrent.atomic.AtomicInteger add_count = new java.util.concurrent.atomic.AtomicInteger(0);

        // 添加方块到缓存（按方块所在 chunk 分组）
        // [LMax Fix V43] 缓存上限: FIFO 驱逐最老 chunk 条目 [长期记忆: 016]
        // V42 实测: 跨 chunk 树半边方块落在永不重载的区块(甜甜圈) → 2930 万方块孤儿撑到 5GB。
        // tracked_keys(ConcurrentHashMap.newKeySet, size() 低开销) 去重登记, insertion_order 记 FIFO 序。
        // 不变量: tracked_keys 与 insertion_order 1:1(add 双写, place/placeForced 双清, evict 双删)。
        private static final java.util.concurrent.ConcurrentLinkedQueue<ChunkPos> insertion_order = new java.util.concurrent.ConcurrentLinkedQueue<>();
        private static final java.util.Set<ChunkPos> tracked_keys = java.util.concurrent.ConcurrentHashMap.newKeySet();
        // [LMax Fix V44.3] 维度路由表(修2b): chunk -> 所属维度. 与 tracked_keys/insertion_order 同生命周期
        // (add 登记 / place+evict 双清), 供低水位回灌按维找 ServerLevel. [长期记忆: 023]
        private static final java.util.Map<ChunkPos, net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level>> chunk_dims
                = new java.util.concurrent.ConcurrentHashMap<>();

        // [LMax Fix V43] FIFO 驱逐: 摘队头键 → 清记账 → 删缓存条目。被驱 chunk 已有方块时走 log_queue_overflow 键控告警。
        private static void evictOldestChunk () {
            ChunkPos oldest = insertion_order.poll();
            if (oldest == null) return;
            tracked_keys.remove(oldest);
            chunk_dims.remove(oldest); // [LMax Fix V44.3] 维度路由记账同步驱逐(修2b)
            Map<BlockPos, BlockState> evicted = cache_blocks.remove(oldest);
            if (evicted != null && evicted.isEmpty() == false) {
                if (Core.log_queue_overflow) System.err.println("[TansHugeTrees] PendingBlocks overflow, evicted chunk " + oldest + " (" + evicted.size() + " blocks)");
            }
        }
        // [LMax Fix V44.3] 签名加 level_server: 新键登记维度路由 chunk_dims(修2b 回灌按 ResourceKey 找
        // ServerLevel, 防跨维度 ChunkPos 撞车误写——纯 ChunkPos 键的多维歧义是既有债, 此处补消费侧路由)
        private static void add (net.minecraft.server.level.ServerLevel level_server, BlockPos pos, BlockState block) {
            ChunkPos chunk_pos = new ChunkPos(pos);
            // [LMax Fix V43] 新键登记 + 溢出驱逐(先驱逐后写入, 保证写入后 ≤ pending_blocks_max_chunks;
            // isEmpty 卫语句防极端竞态下队列空转死循环)
            if (tracked_keys.add(chunk_pos)) {
                insertion_order.offer(chunk_pos);
                chunk_dims.put(chunk_pos, level_server.dimension()); // [LMax Fix V44.3] 维度路由登记
                while (insertion_order.isEmpty() == false && tracked_keys.size() > Handcode.Config.pending_blocks_max_chunks) {
                    evictOldestChunk();
                }
            }
            cache_blocks.computeIfAbsent(chunk_pos, create -> new java.util.concurrent.ConcurrentHashMap<>()).put(pos, block);
            // [LMax Debug V39] 诊断：追踪方块入缓存总数
            if (Core.log_pending_blocks) {
                long total = add_count.incrementAndGet();
                if (total % 100 == 0) System.out.println("[THT-DEBUG] PendingBlocks.add() total: " + total + " blocks, last target chunk: " + chunk_pos);
            }
        }

        // 从缓存中拉取并写入指定 chunk 的所有方块（由各 chunk 的 start() / 重载 flush / 低水位回灌调用）
        // [LMax Fix V44] int→void + 改投 PlacementQueue: 原直落循环在调用线程(=TREE_GEN_EXECUTOR)上
        // 直接写世界, 属于"executor 碰 Level"违例(V43c 死锁根因族); 现在原子摘取后入队, 主线程滴灌. [长期记忆: 018]
        // [LMax Fix V44.3] 修1 三门 + 吸收 placeForced(单一入口): instanceof 门天然分流 proto/Load 两路径
        // (proto 拒绝摘缓存, 留待升格 FULL 后 Load 冲刷); 返回 boolean 供低水位回灌计数(修2b). [长期记忆: 023]
        private static boolean place (LevelAccessor level_accessor, ChunkPos chunk_pos) {

            // [LMax Fix V44.3] 修1 门1 proto 分流器: worldgen 管线(Feature 钩子)传入的 level_accessor 是
            // WorldGenRegion——V44 旧注释"调用方均为 ServerLevel"不变量漏数了 WorldGenStepBeforePlants
            // 调用方, proto 路径跑到旧 cast = ClassCastException 直炸 vanilla 生成管线. instanceof 门=留缓存零风险
            if ((level_accessor instanceof net.minecraft.server.level.ServerLevel) == false) {
                return false;
            }
            net.minecraft.server.level.ServerLevel level_server = (net.minecraft.server.level.ServerLevel) level_accessor;

            // [LMax Fix V44.3] 修1 门2 FULL 状态门: getChunkNow 纯缓存查询(未加载/proto 升格中均 null 零等待);
            // 非 FULL 时 Tile.set 内部 getChunk(FULL) 同步等升格(60 次 50-500ms stall 机理), 拒绝摘缓存,
            // 留待 proto→FULL 升格触发的 ChunkEvent.Load 再冲刷(二次唤醒闭环, 与 V43 挂起机制同语义)
            net.minecraft.world.level.chunk.ChunkAccess chunk_now =
                    level_server.getChunkSource().getChunkNow(chunk_pos.x, chunk_pos.z);
            if (chunk_now == null || !chunk_now.getHighestGeneratedStatus().isOrAfter(net.minecraft.world.level.chunk.ChunkStatus.FULL)) {
                return false;
            }

            // [LMax Fix V44.3] 修1 门3 容量门: 队列饱和 = 消费端暂态落后, 摘走即无副本(drop 已废止);
            // 拒绝摘缓存由修2b 低水位回灌自愈, 数据零丢失
            if (PlacementQueue.isSaturated()) {
                return false;
            }

            // [LMax Fix V44] get→remove 原子摘取(所有权转移给队列): 并发调用天然幂等, 后到者拿 null 零开销;
            // FIFO 记账同步出队(tracked_keys 与 insertion_order 1:1 维护)
            Map<BlockPos, BlockState> data = cache_blocks.remove(chunk_pos);
            tracked_keys.remove(chunk_pos);
            insertion_order.remove(chunk_pos);
            chunk_dims.remove(chunk_pos); // [LMax Fix V44.3] 维度路由记账同步出队(修2b)

            // [LMax Debug V39] 诊断：place()调用时缓存状态
            if (Core.log_pending_blocks) {
                System.out.println("[THT-DEBUG] PendingBlocks.place() chunk " + chunk_pos + " cache: " + (data == null ? "NULL" : data.size() + " blocks") + " | total cache chunks: " + cache_blocks.size());
            }

            if (data == null) {
                return true; // 空缓存=幂等无事可做, 计成功供回灌推进(记账已清)
            }

            // [LMax Fix V44.3] 落块策略: Tile.set worldgen 分支(lc.setBlockState + setUnsaved) 零级联写原语,
            // 仅执行位置从 executor 换到主线程滴灌(原 V44 注释语义保留; 类型不变量修正见门1)
            java.util.function.BiConsumer<BlockPos, BlockState> placer = (pos, block) ->
                    GameUtils.Tile.set(level_server, pos, block, true);

            PlacementQueue.enqueue(level_server, chunk_pos, data, placer,
                    () -> tannyjung.tanshugetrees_core.game.EventCenter.Server.resyncChunk(level_server, chunk_pos));
            return true;
        }

        // [LMax Fix V44.3] 修2b 低水位回灌: 事件路径(Load 冲刷)因容量门拒绝的方块滞留缓存, 队列空闲时
        // 由此自愈重投. 维度经 chunk_dims 路由(防跨维 ChunkPos 撞车误写). 弱一致迭代, 头部 chunk 永不
        // FULL 时预算耗在头部 = 无害(FIFO 语义 + Load 事件路径会先行冲刷). [长期记忆: 023]
        private static void refillTick (net.minecraft.server.MinecraftServer server) {
            if (Handcode.Config.placement_queue_refill_per_tick <= 0) return; // 0 = 禁用回灌
            if (PlacementQueue.isSaturated()) return; // 饱和: 不再投新
            if (PlacementQueue.queueDepth() >= Handcode.Config.placement_queue_refill_low_water) return; // 队列仍忙
            if (insertion_order.isEmpty()) return; // 无缓存: 零开销快退
            int placed = 0;
            java.util.Iterator<ChunkPos> it = insertion_order.iterator(); // 弱一致: 迭代中并发摘除安全
            while (placed < Handcode.Config.placement_queue_refill_per_tick && it.hasNext()) {
                ChunkPos c = it.next();
                net.minecraft.resources.ResourceKey<net.minecraft.world.level.Level> dim_key = chunk_dims.get(c);
                if (dim_key == null) continue; // 记账缺失(竞态窗口): 下 tick 重查
                net.minecraft.server.level.ServerLevel level = server.getLevel(dim_key);
                if (level == null) continue; // 维度未加载: 跳过
                if (place(level, c)) placed++; // 三门拒绝=false 不计数(下 tick 从头部重试)
            }
        }


    }

    private static class Function {

        // [LMax Fix V7] 替换为线程安全容器，解决 ForkJoinPool 并发修改异常
        private static final Map<ChunkPos, Map<BlockPos, List<String>>> cache_functions = new java.util.concurrent.ConcurrentHashMap<>();

        private static void add (ChunkPos chunk_pos, BlockPos pos, String path) {
            cache_functions.computeIfAbsent(chunk_pos, create -> new java.util.concurrent.ConcurrentHashMap<>())
                           .computeIfAbsent(pos, create -> new java.util.concurrent.CopyOnWriteArrayList<>())
                           .add(path);
        }

        private static void run (LevelAccessor level_accessor, ServerLevel level_server, ChunkPos chunk_pos) {

            {

                Map<BlockPos, List<String>> data = cache_functions.get(chunk_pos);

                if (data == null) {

                    return;

                }

                for (Map.Entry<BlockPos, List<String>> entry : data.entrySet()) {

                    for (String get : entry.getValue()) {

                        TXTFunction.run(level_accessor, level_server, entry.getKey(), get, false);

                    }

                }

                cache_functions.remove(chunk_pos);

            }

        }

    }

    // [LMax Fix V44] PlacementQueue: 主线程滴灌式方块放置队列(V43c 死锁的结构性修复) [长期记忆: 018]
    // 架构铁律: 一切 Level 写入回主线程, executor 只做纯计算. 本队列即"计算与调度解耦"的外壳:
    //   生产端(任意线程): enqueue 原子摘取方块清单所有权, O(1) 无锁入队
    //   消费端(主线程):   drainTick 按 getBudgetMs() 时间片消费, 主线程开销恒定, 与队列深度无关
    //   策略函数化:       placer lambda 决定"怎么落"(Tile.set 直放/落叶衍生/未来任意), 队列零改动接入
    //   因果时序:         onComplete(=resync) 只在方块全部落地后触发, 杜绝"包比方块先到"幽灵回归
    private static class PlacementQueue {

        // [LMax Fix V44] Job = 最小执行单元: 一个 chunk 的一批方块 + 游标 + 策略 + 完工回调
        private static class Job {
            final net.minecraft.server.level.ServerLevel level;
            final net.minecraft.world.level.ChunkPos chunk_pos;
            final Map.Entry<BlockPos, BlockState>[] blocks; // Map 一次性数组化: 游标推进 O(1), 免 iterator 跨 tick 状态
            int cursor = 0; // [LMax Fix V44] 跨 tick 续放游标(滴灌的本质: 时间片用尽即暂停, 下 tick 从游标续)
            final java.util.function.BiConsumer<BlockPos, BlockState> placer;
            final Runnable onComplete;

            @SuppressWarnings("unchecked")
            Job (net.minecraft.server.level.ServerLevel level, net.minecraft.world.level.ChunkPos chunk_pos, Map<BlockPos, BlockState> blocks, java.util.function.BiConsumer<BlockPos, BlockState> placer, Runnable onComplete) {
                this.level = level;
                this.chunk_pos = chunk_pos;
                this.blocks = blocks.entrySet().toArray(new Map.Entry[0]); // [LMax Fix V44] 数组化: 消费期只读
                this.placer = placer;
                this.onComplete = onComplete;
            }
        }

        // [LMax Fix V44] 无界并发队列: 深度有界由三个 place() 入口的 isSaturated() 容量门 + canary 告警兜底
        // (V44.3 修2a: 旧"enqueue 背压丢弃"废止——见下)
        private static final java.util.concurrent.ConcurrentLinkedQueue<Job> queue = new java.util.concurrent.ConcurrentLinkedQueue<>();

        // [LMax Fix V44] 生产端: 任意线程 O(1). 空 map 直接完工(不发空 job).
        // [LMax Fix V44.3] 修2a: 旧"触达上限丢最老 job"废止——job 已从缓存原子摘出(所有权转移),
        // 丢弃 = 方块无任何副本必丢(与 V43 PendingBlocks FIFO 驱逐不同: 那时缓存还有账可丢).
        // 生产背压改由三个 place() 入口的 isSaturated() 容量门承担(拒绝摘缓存, 数据零丢失);
        // 本处仅 canary 告警. 队列深度可短暂超上限(新增来源已被门封), drainTick 恒速消化自愈. [长期记忆: 023]
        private static void enqueue (net.minecraft.server.level.ServerLevel level, net.minecraft.world.level.ChunkPos chunk_pos, Map<BlockPos, BlockState> blocks, java.util.function.BiConsumer<BlockPos, BlockState> placer, Runnable onComplete) {

            if (blocks.isEmpty()) {
                onComplete.run(); // [LMax Fix V44] 空批=立即完工(幂等语义, 无方块可落)
                return;
            }

            // [LMax Fix V44.3] canary: 饱和告警不丢弃(投递照常), 触达=消费暂态落后
            if (queue.size() >= Handcode.Config.placement_queue_max_jobs) {
                System.err.println("[TansHugeTrees] PlacementQueue saturated (depth=" + queue.size() + "), accepting job chunk=" + chunk_pos + " (canary, no drop)");
            }

            queue.offer(new Job(level, chunk_pos, blocks, placer, onComplete));
        }

        // [LMax Fix V44.3] 修2a: 饱和判定——三个 place() 入口的容量门共用(拒绝摘缓存=数据滞留缓存零丢失)
        public static boolean isSaturated () {
            return queue.size() >= Handcode.Config.placement_queue_max_jobs;
        }

        // [LMax Fix V44.3] 修2b: 队列深度(低水位回灌判据)
        public static int queueDepth () {
            return queue.size();
        }

        // [LMax Fix V44] 消费端: 仅主线程调用(processTick 尾部挂点). 时间片 = getBudgetMs().
        // 卫兵: 滴灌中 chunk 卸载 → job 回队尾等重载(恢复后由下轮 drainTick 续放;
        // 缓存已被原子摘走, 重载路径 flushPendingBlocks 拿 null 零开销, 与本 job 游标续放互补不冲突);
        // 整圈全部未加载 → break 防空转(下 tick 再试). budget<=0 = 暂停滴灌(任务滞留至上限).
        // [LMax Fix V44.3] 卫兵升级 hasChunk→getChunkNow+FULL: hasChunk 对已入 loaded 表的 proto chunk
        // 也返回 true, 放行后 Tile.set 内部 getChunk(FULL) 同步等 proto 升格 = 主线程 stall
        // (V44 实测 60 次 50-500ms mod stall 机理); getChunkNow 纯缓存查询, proto 升格中返回 null
        // 零等待, job 回队尾等 FULL 提升后再续放. [长期记忆: 023]
        public static void drainTick () {

            if (queue.isEmpty()) return; // [LMax Fix V44] 空队列零开销快退

            long deadline = System.nanoTime() + getBudgetMs() * 1_000_000L;
            int requeued = 0;

            Job job;
            while (System.nanoTime() < deadline && (job = queue.poll()) != null) {

                net.minecraft.world.level.chunk.ChunkAccess job_chunk =
                        job.level.getChunkSource().getChunkNow(job.chunk_pos.x, job.chunk_pos.z);
                if (job_chunk == null || !job_chunk.getHighestGeneratedStatus().isOrAfter(net.minecraft.world.level.chunk.ChunkStatus.FULL)) {
                    queue.offer(job); // 回队尾等待 FULL(卸载或 proto 升格中同判)
                    requeued++;
                    if (requeued >= queue.size()) break; // [LMax Fix V44] 整圈未加载: 防空转
                    continue;
                }

                try {
                    while (job.cursor < job.blocks.length && System.nanoTime() < deadline) {
                        Map.Entry<BlockPos, BlockState> entry = job.blocks[job.cursor];
                        job.cursor++;
                        job.placer.accept(entry.getKey(), entry.getValue());
                    }

                    if (job.cursor >= job.blocks.length) {
                        job.onComplete.run(); // [LMax Fix V44] 完工: 方块全部落地后才触发 resync
                    } else {
                        queue.offer(job); // [LMax Fix V44] 时间片用尽: 续放(游标语义)
                    }
                } catch (Throwable t) {
                    // [LMax Fix V44] 消费侧异常围栏: job 级隔离, 单 job 故障不冻结滴灌循环
                    // (游标前已落方块保留, 游标后丢弃; 与 Future 黑洞不同, 此处异常可见)
                    System.err.println("[TansHugeTrees] PlacementQueue job crashed, dropped chunk=" + job.chunk_pos + " (cursor=" + job.cursor + "/" + job.blocks.length + "):");
                    t.printStackTrace();
                }
            }
        }

        // [LMax Fix V44] 自适应预算钩子: 当前直读配置键 placement_queue_budget_ms;
        // 未来接 AIMD 控制器(测量 MSPT 加性增/乘性减动态调节)时只改本方法体, 调度代码零改动
        private static int getBudgetMs () {
            return Handcode.Config.placement_queue_budget_ms;
        }
    }
